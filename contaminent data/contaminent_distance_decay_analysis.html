Overview

This report creates a dummy tissue dataset for sampled individuals across latitude and longitude, then evaluates distance-decay relationships for:

  • DDT
  • PCP
  • Isotope signature (delta13C and delta15N combined)
  • Overall multivariate similarity

In the spirit of broad spatial-scaling analyses, we compare multiple distance-decay functions (exponential, power-law, and log-linear), evaluate isolated curves per variable, and assess generality with a pooled model and null randomization.

2) Pairwise Distance and Similarity

# Haversine distance in km.
haversine_km <- function(lat1, lon1, lat2, lon2) {
  r <- 6371
  to_rad <- pi / 180
  dlat <- (lat2 - lat1) * to_rad
  dlon <- (lon2 - lon1) * to_rad
  a <- sin(dlat / 2)^2 + cos(lat1 * to_rad) * cos(lat2 * to_rad) * sin(dlon / 2)^2
  2 * r * atan2(sqrt(a), sqrt(1 - a))
}

# Convert a distance in variable space into a 0-1 similarity.
sim_from_diff <- function(x, scale_val) {
  exp(-abs(x) / scale_val)
}

n <- nrow(dummy_data)
pairs <- vector("list", n * (n - 1) / 2)
k <- 1

sd_DDT <- sd(dummy_data$DDT)
sd_PCP <- sd(dummy_data$PCP)
sd_iso <- sd(c(dummy_data$delta13C, dummy_data$delta15N))

for (i in 1:(n - 1)) {
  for (j in (i + 1):n) {
    d_geo <- haversine_km(dummy_data$latitude[i], dummy_data$longitude[i],
                          dummy_data$latitude[j], dummy_data$longitude[j])

    d_ddt <- dummy_data$DDT[i] - dummy_data$DDT[j]
    d_pcp <- dummy_data$PCP[i] - dummy_data$PCP[j]
    d_iso <- sqrt((dummy_data$delta13C[i] - dummy_data$delta13C[j])^2 +
                  (dummy_data$delta15N[i] - dummy_data$delta15N[j])^2)

    sim_ddt <- sim_from_diff(d_ddt, sd_DDT)
    sim_pcp <- sim_from_diff(d_pcp, sd_PCP)
    sim_iso <- exp(-d_iso / sd_iso)
    sim_all <- mean(c(sim_ddt, sim_pcp, sim_iso))

    pairs[[k]] <- data.frame(
      site_i = dummy_data$site_id[i],
      site_j = dummy_data$site_id[j],
      distance_km = d_geo,
      sim_DDT = sim_ddt,
      sim_PCP = sim_pcp,
      sim_isotope = sim_iso,
      sim_overall = sim_all,
      stringsAsFactors = FALSE
    )
    k <- k + 1
  }
}

3) Fit Distance-Decay Curves (Isolated + Overall)

fit_decay_models <- function(distance, similarity) {
  dat <- data.frame(distance = pmax(distance, 0), sim = pmin(pmax(similarity, 1e-6), 0.999999))

  # Exponential: sim = a * exp(-b * d)
  exp_fit <- try(nls(sim ~ a * exp(-b * distance),
                     data = dat,
                     start = list(a = 0.9, b = 0.001),
                     control = nls.control(maxiter = 200)),
                 silent = TRUE)

  # Power-law: sim = a * (d + 1)^(-b)
  pow_fit <- try(nls(sim ~ a * (distance + 1)^(-b),
                     data = dat,
                     start = list(a = 0.9, b = 0.1),
                     control = nls.control(maxiter = 200)),
                 silent = TRUE)

  # Log-linear: sim = a + b * log(d + 1)
  loglin_fit <- lm(sim ~ log(distance + 1), data = dat)

  get_row <- function(name, fit_obj, type) {
    if (inherits(fit_obj, "try-error")) {
      return(data.frame(model = name, model_type = type, intercept = NA, slope = NA, AIC = NA, stringsAsFactors = FALSE))
    }
    if (type == "lm") {
      co <- coef(fit_obj)
      return(data.frame(
        model = name,
        model_type = type,
        intercept = unname(co[1]),
        slope = unname(co[2]),
        AIC = AIC(fit_obj),
        stringsAsFactors = FALSE
      ))
    }
    co <- coef(fit_obj)
    data.frame(
      model = name,
      model_type = type,
      intercept = unname(co[1]),
      slope = unname(co[2]),
      AIC = AIC(fit_obj),
      stringsAsFactors = FALSE
    )
  }

  model_table <- rbind(
    get_row("exponential", exp_fit, "nls"),
    get_row("power_law", pow_fit, "nls"),
    get_row("log_linear", loglin_fit, "lm")
  )

  list(models = list(exp = exp_fit, pow = pow_fit, loglin = loglin_fit), table = model_table)
}

4) Null Model (Randomized Tissue Composition)

get_log_slope <- function(distance, similarity) {
  fit <- lm(similarity ~ log(distance + 1))
  unname(coef(fit)[2])
}

plot_decay <- function(distance, similarity, title_txt) {
  plot(distance, similarity,
       pch = 16, cex = 0.35,
       col = rgb(0, 0, 0, 0.18),
       xlab = "Distance (km)", ylab = "Similarity",
       main = title_txt)

  dseq <- seq(min(distance), max(distance), length.out = 300)

  fobj <- fit_decay_models(distance, similarity)$models

  if (!inherits(fobj$exp, "try-error")) {
    ce <- coef(fobj$exp)
    lines(dseq, ce[1] * exp(-ce[2] * dseq), lwd = 2, col = "firebrick")
  }
  if (!inherits(fobj$pow, "try-error")) {
    cp <- coef(fobj$pow)
    lines(dseq, cp[1] * (dseq + 1)^(-cp[2]), lwd = 2, col = "darkorange3")
  }
  cl <- coef(fobj$loglin)
  lines(dseq, cl[1] + cl[2] * log(dseq + 1), lwd = 2, col = "steelblue4")

  legend("topright", bty = "n", cex = 0.8,
         legend = c("Exponential", "Power-law", "Log-linear"),
         col = c("firebrick", "darkorange3", "steelblue4"), lwd = 2)
}

  # Compute per-bin mean ± 95% CI (t-based).
  bin_sim <- function(df, var, n_bins = 12) {
    df <- df[is.finite(df[[var]]) & is.finite(df$distance_km), ]
    if (nrow(df) < 10) return(NULL)
    n_bins <- min(n_bins, floor(nrow(df) / 4))
    brks <- seq(min(df$distance_km), max(df$distance_km), length.out = n_bins + 1)
    df$bin <- cut(df$distance_km, breaks = brks, labels = FALSE, include.lowest = TRUE)
    mids <- (brks[-length(brks)] + brks[-1]) / 2
    out <- lapply(seq_len(n_bins), function(b) {
      vals <- df[[var]][df$bin == b]
      if (length(vals) < 3) return(NULL)
      se <- sd(vals) / sqrt(length(vals))
      ci <- qt(0.975, df = length(vals) - 1) * se
      data.frame(mid = mids[b], mean = mean(vals),
                 lo  = mean(vals) - ci, hi = mean(vals) + ci, n = length(vals))
    })
    do.call(rbind, Filter(Negate(is.null), out))
  }

  plot_binned <- function(bd, ylab, title_txt, col_line = "firebrick") {
    if (is.null(bd) || nrow(bd) < 2) { plot.new(); title(title_txt); return(invisible()) }
    ylim_r <- c(pmax(0, min(bd$lo, na.rm = TRUE) - 0.04),
                pmin(1, max(bd$hi, na.rm = TRUE) + 0.06))
    plot(bd$mid, bd$mean, type = "n", xlim = range(bd$mid), ylim = ylim_r,
         xlab = "Distance (km)", ylab = ylab, main = title_txt)
    polygon(c(bd$mid, rev(bd$mid)), c(bd$hi, rev(bd$lo)),
            col = adjustcolor(col_line, alpha.f = 0.18), border = NA)
    lines(bd$mid, bd$mean, lwd = 2.5, col = col_line)
    points(bd$mid, bd$mean, pch = 21, bg = "white", col = col_line, cex = 1.3, lwd = 1.8)
    text(bd$mid, bd$hi + diff(ylim_r) * 0.04,
         labels = paste0("n=", bd$n), cex = 0.52, srt = 45, adj = 0)
  }

  par(mfrow = c(2, 2), mar = c(4, 4, 3, 1))
  plot_binned(bin_sim(pair_real, "sim_DDT"),        "Mean Similarity (DDT)",     "DDT – Binned Mean ± 95% CI (all species)",     "firebrick")

13) Interpretation and Discussion

Overview

This analysis characterised spatial distance-decay in chemical similarity across 23220 pairwise comparisons drawn from seabird tissue samples collected across Alaskan and North Pacific geographic areas. Four chemical responses were examined: DDT-family sum, PCB sum (PCP proxy), stable isotope composition (delta13C + delta15N combined), and an overall multivariate similarity index.

DDT: Strong, Exponentially Structured Spatial Decay

DDT-family similarity showed the strongest and most spatially structured distance-decay of any response (observed log-linear slope = -0.0182, null mean = -0.0024, p < 0.002). The exponential model substantially outperformed power-law and log-linear alternatives (ΔAIC > 1784), with the fitted form:

sim = 0.6457 × exp(−2.54 × 10⁻⁴ × d)

This implies rapid initial similarity loss at short distances — consistent with point-source pollution that attenuates with distance from coastal urbanisation and agricultural watersheds. Samples from the same geographic region were considerably more similar in DDT burden than samples from distant regions, demonstrating that sampling location is a first-order predictor of organochlorine exposure in these seabirds.

PCP / PCB: Weaker but Statistically Significant Decay

PCB-based similarity showed a weaker signal (slope = -0.0082, null = 0.0022, p < 0.002), suggesting that PCB contamination is more spatially diffuse. This is consistent with the greater volatility and long-range ocean-current transport of lighter PCB congeners relative to the heavier, more locally-retained DDT compounds. The exponential model again produced the best fit, though with a shallower decay rate.

Isotopes: Log-linear Decay Reflecting Foraging Area Structure

δ¹³C + δ¹⁵N similarity followed a log-linear decay most closely (best AIC = −8674, vs. −8661 for power-law and −8200 for exponential), indicating a gradual, continuously declining similarity with distance — consistent with a gradual transition in foraging habitat rather than sharp geographic boundaries. The observed slope (-0.0228) was significantly more negative than the spatially-constrained null (-0.017), confirming that isotopic signatures track broad-scale foraging zones.

Overall Multivariate Similarity

The four-variable pooled similarity index (slope = -0.0164) integrates contaminant and isotopic signals and recovers the strongest model support (best AIC = −14,431 for exponential), reflecting that multivariate distance-decay is more coherently structured than any single variable alone.

Species-Level Variation

Uria aalge and Uria lomvia contribute the majority of within-species pairs (> 12,500 and > 5,500 respectively). Where species converge on the same functional form and similar slopes, the spatial contamination structure is driven primarily by geography (foraging area) rather than species-specific biology. Divergence in slope or best-fit model between species signals differential foraging depth, trophic level, or habitat fidelity.

Limitations

  • Spatial confounding: samples cluster by geographic area; within-area pairs have distance = 0, anchoring high-similarity values at d = 0.
  • Non-independence: n(n−1)/2 pairwise comparisons are not independent — the null model permutation test accounts for this partially.
  • Unit harmonisation: DDT and PCB values retain units from the modal unit per analyte group; cross-study mismatches would inflate apparent dissimilarity.
  • Missing predictive covariates: oceanographic variables (SST, chlorophyll, bathymetry), diet, age, and sex are not included but are essential for fuller mechanistic models.

14) Predictive Modeling Extensions

What Does a Contamination Modeler Need?

A researcher interested in predicting contamination at unsampled sites or future time points would typically seek:

  1. Spatial autocorrelation structure (variogram / distance-decay) — done above — defines the range at which samples are spatially informative.
  2. Kriging or spatial interpolation — predicts concentration at unsampled coordinates weighted by the empirical variogram.
  3. GAM / GLMM spatial prediction — models concentration as a smooth function of latitude, longitude, species, and year; allows extrapolation.
  4. Spatial cross-validation — leave-one-area-out (LOAO) or block holdout CV gives unbiased prediction error; standard k-fold CV is anti-conservative for spatially autocorrelated data.
  5. Temporal trend modeling — tests whether concentrations are declining (regulation effect) or increasing (new sources).
  6. Environmental covariate linkage — SST, chlorophyll, bathymetry as predictors for transfer to climate scenarios.

Sections 14A–14C implement items 3, 4, and 5 using the individual-level ZD Stamp sample data.

15) Spatiotemporal Scaling Model (Distance + Time)

This section fits a joint scaling model for contaminant similarity as a function of both geographic distance and collection-year separation:

\[ ext{sim}_{ij} = \beta_0 + \beta_1\log(d_{ij}+1) + \beta_2\log(\Delta t_{ij}+1) + \beta_3\log(d_{ij}+1)\log(\Delta t_{ij}+1) + \varepsilon_{ij} \]

where \(d_{ij}\) is pairwise geographic distance (km) and \(\Delta t_{ij}\) is pairwise absolute time lag (years).

15B) Dyadic Mixed-Effects Spatiotemporal Model + Blocked CV

To reduce pseudo-replication from pairwise expansion, this section fits a dyadic mixed-effects model with crossed random effects for both members of each pair (individual IDs, species, and geographic areas), then evaluates prediction using leave-one-area-out blocked cross-validation.

if (!exists("pair_real")) {
  cat("pair_real not available — skipping dyadic mixed-effects model.\n")
} else if (!all(c("time_lag_years", "id_i", "id_j", "area_i", "area_j", "species_i", "species_j") %in% names(pair_real))) {
  cat("Required pairwise fields not found — rerun real-data analysis chunk first.\n")

  if (!requireNamespace("lme4", quietly = TRUE)) {
    install.packages("lme4", repos = "https://cran.r-project.org/")
  }
  library(lme4)

  fit_dyadic <- function(df, yvar) {
    keep <- is.finite(df[[yvar]]) &
      is.finite(df$distance_km) & is.finite(df$time_lag_years) &
      !is.na(df$id_i) & !is.na(df$id_j) & !is.na(df$area_i) & !is.na(df$area_j) &
      !is.na(df$species_i) & !is.na(df$species_j)

    dat <- df[keep, c(yvar, "distance_km", "time_lag_years", "id_i", "id_j", "area_i", "area_j", "species_i", "species_j")]
    names(dat)[1] <- "y"

    if (nrow(dat) < 100) return(NULL)

    dat$log_dist <- log(dat$distance_km + 1)
    dat$log_time <- log(dat$time_lag_years + 1)

    dat$id_i <- as.factor(dat$id_i)
    dat$id_j <- as.factor(dat$id_j)
    dat$area_i <- as.factor(dat$area_i)
    dat$area_j <- as.factor(dat$area_j)
    dat$species_i <- as.factor(dat$species_i)
    dat$species_j <- as.factor(dat$species_j)

    ctrl <- lmerControl(optimizer = "bobyqa", optCtrl = list(maxfun = 1e5))

    full_formula <- y ~ log_dist * log_time +
      (1 | id_i) + (1 | id_j) +
      (1 | species_i) + (1 | species_j) +
      (1 | area_i) + (1 | area_j)

    fallback_formula <- y ~ log_dist * log_time +
      (1 | species_i) + (1 | species_j) +
      (1 | area_i) + (1 | area_j)

    mod <- try(lmer(full_formula, data = dat, REML = FALSE, control = ctrl), silent = TRUE)
    model_type <- "full_dyadic"
    if (inherits(mod, "try-error")) {
      mod <- try(lmer(fallback_formula, data = dat, REML = FALSE, control = ctrl), silent = TRUE)
      model_type <- "fallback_species_area"
    }
    if (inherits(mod, "try-error")) return(NULL)

    sm <- summary(mod)
    cf <- as.data.frame(sm$coefficients)
    cf$term <- rownames(cf)

    get_cf <- function(term, colname) {
      if (!(term %in% cf$term)) return(NA_real_)
      cf[cf$term == term, colname][1]
    }

    fixed_row <- data.frame(
      response = yvar,
      model_type = model_type,
      n_pairs = nrow(dat),
      beta_dist = get_cf("log_dist", "Estimate"),
      beta_time = get_cf("log_time", "Estimate"),
      beta_dist_time = get_cf("log_dist:log_time", "Estimate"),
      se_dist = get_cf("log_dist", "Std. Error"),
      se_time = get_cf("log_time", "Std. Error"),
      se_dist_time = get_cf("log_dist:log_time", "Std. Error"),
      t_dist = get_cf("log_dist", "t value"),
      t_time = get_cf("log_time", "t value"),
      t_dist_time = get_cf("log_dist:log_time", "t value"),
      AIC = AIC(mod),
      stringsAsFactors = FALSE
    )

    list(model = mod, data = dat, fixed = fixed_row)
  }

  run_blocked_cv <- function(dat, model_type, yvar) {
    areas <- sort(unique(c(as.character(dat$area_i), as.character(dat$area_j))))
    areas <- areas[!is.na(areas) & nchar(trimws(areas)) > 0]
    if (length(areas) < 3) return(NULL)

    ctrl <- lmerControl(optimizer = "bobyqa", optCtrl = list(maxfun = 1e5))
    full_formula <- y ~ log_dist * log_time +
      (1 | id_i) + (1 | id_j) +
      (1 | species_i) + (1 | species_j) +
      (1 | area_i) + (1 | area_j)
    fallback_formula <- y ~ log_dist * log_time +
      (1 | species_i) + (1 | species_j) +
      (1 | area_i) + (1 | area_j)

    cv_rows <- lapply(areas, function(a) {
      test_idx <- as.character(dat$area_i) == a | as.character(dat$area_j) == a
      train <- dat[!test_idx, ]
      test <- dat[test_idx, ]
      if (nrow(train) < 100 || nrow(test) < 20) return(NULL)

      form <- if (model_type == "full_dyadic") full_formula else fallback_formula
      fit <- try(lmer(form, data = train, REML = FALSE, control = ctrl), silent = TRUE)
      if (inherits(fit, "try-error")) {
        fit <- try(lmer(fallback_formula, data = train, REML = FALSE, control = ctrl), silent = TRUE)
      }
      if (inherits(fit, "try-error")) return(NULL)

      pred <- try(predict(fit, newdata = test, allow.new.levels = TRUE), silent = TRUE)
      if (inherits(pred, "try-error")) return(NULL)

      obs <- test$y
      rmse <- sqrt(mean((obs - pred)^2, na.rm = TRUE))
      mae <- mean(abs(obs - pred), na.rm = TRUE)
      r2 <- if (length(unique(obs)) > 1) cor(obs, pred, use = "complete.obs")^2 else NA_real_

      data.frame(
        response = yvar,
        held_out_area = a,
        n_test = nrow(test),
        RMSE = rmse,
        MAE = mae,
        R2_pred = r2,
        stringsAsFactors = FALSE
      )
    })

    cv_tab <- do.call(rbind, Filter(Negate(is.null), cv_rows))
    if (is.null(cv_tab) || nrow(cv_tab) == 0) return(NULL)
    mean_row <- data.frame(
      response = yvar,
      held_out_area = "Mean across held-out areas",
      n_test = NA,
      RMSE = mean(cv_tab$RMSE, na.rm = TRUE),
      MAE = mean(cv_tab$MAE, na.rm = TRUE),
      R2_pred = mean(cv_tab$R2_pred, na.rm = TRUE),
      stringsAsFactors = FALSE
    )
    rbind(cv_tab, mean_row)
  }

  targets <- c("sim_overall", "sim_DDT")
  fit_objs <- lapply(targets, function(v) fit_dyadic(pair_real, v))
  names(fit_objs) <- targets
  fit_objs <- fit_objs[!vapply(fit_objs, is.null, logical(1))]

  if (length(fit_objs) == 0) {
    cat("Dyadic mixed-effects models failed or insufficient data.\n")
  } else {

    write.csv(fixed_tab, file.path(out_dir, "dyadic_mixed_model_fixed_effects.csv"), row.names = FALSE)

    print(knitr::kable(
      fixed_tab,
      row.names = FALSE,
      col.names = c(
        "Response", "Model", "N pairs",
        "Beta distance", "Beta time", "Beta distance x time",
        "SE dist", "SE time", "SE dist x time",
        "t dist", "t time", "t dist x time", "AIC"

      out

    if (!is.null(cv_tabs) && nrow(cv_tabs) > 0) {
      cv_tabs$response <- c(sim_overall = "Overall", sim_DDT = "DDT")[cv_tabs$response]
      num_cols <- c("n_test", "RMSE", "MAE", "R2_pred")
      for (nc in num_cols) {
        if (nc %in% names(cv_tabs)) cv_tabs[[nc]] <- round(cv_tabs[[nc]], 4)

      write.csv(cv_tabs, file.path(out_dir, "dyadic_mixed_model_blocked_cv.csv"), row.names = FALSE)

        cv_tabs,
        row.names = FALSE,

    fit_dyadic_trophic <- function(df, yvar) {
      keep <- is.finite(df[[yvar]]) &
        is.finite(df$distance_km) & is.finite(df$time_lag_years) &
        is.finite(df$trophic_diff_n15) & is.finite(df$habitat_diff_c13) &
        is.finite(df$mean_trophic_n15) & is.finite(df$mean_habitat_c13) &
        !is.na(df$id_i) & !is.na(df$id_j) & !is.na(df$area_i) & !is.na(df$area_j) &
        !is.na(df$species_i) & !is.na(df$species_j)

        yvar, "distance_km", "time_lag_years",
        "trophic_diff_n15", "habitat_diff_c13", "mean_trophic_n15", "mean_habitat_c13",
        "id_i", "id_j", "area_i", "area_j", "species_i", "species_j"

      dat$id_j <- as.factor(dat$id_j)
      dat$area_i <- as.factor(dat$area_i)

      dat$species_j <- as.factor(dat$species_j)

        (1 | area_i) + (1 | area_j)

        (1 | species_i) + (1 | species_j) +
        (1 | area_i) + (1 | area_j)

        beta_mean_trophic = get_cf("mean_trophic_n15", "Estimate"),
        beta_mean_habitat = get_cf("mean_habitat_c13", "Estimate"),

        stringsAsFactors = FALSE
      )

    }

      areas <- sort(unique(c(as.character(dat$area_i), as.character(dat$area_j))))

16) Detailed Methods and Hypothesis-Testing Framework

This section formalizes the analytical strategy as a scale-aware hypothesis-testing workflow. The goal is not only to estimate distance-decay, but to test whether observed patterns exceed what would be expected under ecologically informed null structure.

16A. Core inferential setup

The analysis is built around pairwise comparisons among samples. For each pair, we quantify:

  • geographic separation (km),
  • contaminant similarity (DDT family; PCP/PCB family),
  • isotope similarity (delta13C and delta15N structure), and
  • multivariate overall similarity.

This pairwise framing makes scale explicit because each observation is a contrast across space (and later time), not a single-site value.

16B. Hypotheses and matched tests

The workflow tests the following hypotheses.

  1. H1: Spatial decay exists for each response. Test: fit exponential, power-law, and log-linear distance-decay functions to each response and evaluate slope direction, fit quality, and AIC support.

  2. H2: Observed decay is stronger than expected under structured randomness. Test: permutation null models comparing observed log-distance slopes to null slope distributions (p_lower and p_upper). This is the principal significance test for distance structure.

  3. H3: Decay form differs by contaminant class. Test: compare functional-form rankings and parameter values among DDT-family, PCP/PCB-family, isotope, and overall similarity responses.

  4. H4: Distance-decay is not species-invariant. Test: species-level model fits and slope summaries, followed by cross-species synthesis table of best models and decay strength.

  5. H5: Time contributes independent structure beyond distance. Test: spatiotemporal scaling models including distance, temporal lag, and distance-by-time interaction terms.

  6. H6: Hierarchical dependence matters for inference and prediction. Test: dyadic mixed-effects models with crossed random effects for individuals, species, and areas, preventing pseudoreplication from pairwise data reuse.

  7. H7: Biodiversity-relevant trophic structure improves transferability. Test: extension of dyadic mixed models with trophic and habitat isotope covariates, compared against baseline models under blocked transfer tests.

16C. Null models as a central inferential engine

Null models are not a side diagnostic in this workflow; they are the backbone of hypothesis testing for spatial structure.

  • In the synthetic demonstration, randomization establishes the baseline slope distribution expected when spatial signal is absent.
  • In real data, spatially constrained permutations preserve broad area and species composition while breaking fine-scale pairwise alignment, producing a conservative counterfactual for H2.

This design answers the key inferential question directly: is the observed decay stronger than expected given the sampling structure and broad ecological grouping?

16D. Scale as the design principle

Scale is explicitly embedded in each stage:

  • Fine-to-broad spatial scale: pairwise distances from local to basin-wide extents, with flexible functional forms to detect curvature and attenuation behavior.
  • Temporal scale: year and pairwise lag terms test whether spatial decay is stable or shifts with time separation.
  • Biological scale: species-level and trophic structure test whether ecological organization changes decay strength or model transfer performance.
  • Prediction scale: leave-one-area-out blocking evaluates extrapolation to truly new spatial domains rather than interpolative fit alone.

Together, these choices convert scale from a descriptive backdrop into an explicit driver of model architecture and hypothesis tests.

16E. Why this multi-method sequence is necessary

No single model can answer all questions in contamination biogeography. The sequence used here is deliberate:

  1. identify form and strength of decay,
  2. validate against null expectations,
  3. evaluate heterogeneity among species and chemistry classes,
  4. incorporate temporal coupling,
  5. model hierarchical dependence, and
  6. test ecological covariates for spatial transfer.

This sequence balances mechanistic interpretation and predictive robustness, while keeping inference anchored to explicit null expectations and scale-aware validation.

16F. How trophic level enters the models

Trophic structure is entered through isotope-derived covariates in the dyadic trophic model:

  • trophic_diff_n15 = |delta15N_i - delta15N_j|, a proxy for trophic-level mismatch,
  • habitat_diff_c13 = |delta13C_i - delta13C_j|, a proxy for habitat/foraging mismatch,
  • mean_trophic_n15, the pair-average trophic position proxy,
  • mean_habitat_c13, the pair-average habitat baseline proxy.

Interpretation on the similarity scale is direct: negative coefficients for mismatch terms indicate that isotopically dissimilar pairs are less similar in contaminants after accounting for distance, time lag, and hierarchical random effects.

Important caution: these are ecological proxies, not direct causal measurements of trophic level or habitat identity. delta15N and delta13C each include baseline spatial variation, source mixing, and tissue-integration effects; therefore, coefficients should be interpreted as associational signals unless baseline correction and discrimination-factor uncertainty are modeled explicitly.

16H. Dyadic mixed models explained for undergraduates

A dyadic mixed model is a model for data where each row describes a pair (a dyad), not a single sample. Here, each row compares sample i and sample j.

Why this matters: the same sample appears in many different pairs, so pairwise rows are not independent. If we ignore that, we can get overconfident p-values and unstable effects.

The model solves this by combining:

  • fixed effects (distance, time, trophic covariates): average relationships we want to estimate, and
  • random effects (id_i/id_j, species_i/species_j, area_i/area_j): adjustments that absorb repeated-measure structure and group-level dependence.

In plain language: fixed effects tell us the average rule; random effects acknowledge that each bird/species/area has its own baseline around that rule. Because each pair has two members, modeling both i and j keeps the analysis symmetric; if this is ignored, repeated use of the same samples can make effects look more certain than they really are.

17) Conclusion: Main Drivers of Contamination Variation

Across contaminants and isotopes, the dominant source of variation is space: chemical similarity decreases with increasing geographic distance, and this spatial decay is strongest for DDT-family chemistry and strongest overall in the multivariate index.

The second driver is contaminant class and transport behavior:

  • DDT-family patterns are steep and typically best described by exponential decay, consistent with spatially localized or regionally concentrated sources plus distance attenuation.
  • PCP/PCB patterns are shallower, consistent with broader transport and mixing.
  • Isotopes show smoother, more gradient-like structure (often better log-linear support), indicating broad foraging/habitat transitions rather than sharply bounded sources.

The third driver is time and space-time coupling:

  • Temporal lags contribute additional variation, but usually less than distance.
  • The joint space-time model indicates that distance-decay strength can shift with temporal separation (interaction term), meaning spatial structure is not fully time-stationary.

The fourth driver is species/ecology context:

  • Species-level fits show different slope magnitudes and best functional forms, implying trophic ecology and movement behavior modulate exposure patterns.
  • In this dataset, high-pair species (especially Uria spp.) dominate signal strength, so future inference should balance species representation.

Synthesis of hypothesis support

The strongest empirical support is for H1 and H2: spatial decay is consistently negative, and null-model comparisons indicate that observed decay exceeds structured random expectation. H3 is also supported: DDT-family responses show steeper and often more exponential behavior than PCP/PCB-family and isotope structure. H4 and H6 are supported by species-level heterogeneity and dyadic mixed-model performance under blocked transfer. H5 and H7 are supported but context-dependent: temporal terms and trophic covariates add information, especially in difficult transfer settings, but gains vary by response and held-out region. These trophic gains should be interpreted as improved ecological transfer signals, not stand-alone proof of trophic causation.

Practical take-away for predictive modeling

For prediction, prioritize a spatiotemporal model where contamination is modeled as a function of distance structure, year (or lag), and species effects simultaneously (for example, GAM or mixed-effects spatial model with spatial cross-validation by area). This workflow is best aligned with the evidence in this report: space is the strongest axis, time is informative, and species modifies both.

If a single concise ranking is needed for this report, it is:

  1. Geographic distance (spatial structure)
  2. Contaminant chemistry class (DDT vs PCB/PCP vs isotope behavior)
  3. Time (collection year / pairwise lag)
  4. Species ecological differences
